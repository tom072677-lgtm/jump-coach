# Reliability gates — which numbers are we allowed to say out loud?
#
# The parabola gives ONE equation with TWO unknowns:
#       g_implied = a_px[px/s²] · k² / scale        (k = true fps / container fps)
# so g ≠ 9.81 proves that k or scale (or both) is wrong, but CANNOT say which.
# We therefore never auto-correct scale; we only refuse to quote the numbers that
# depend on it. Angles and times are scale-invariant and survive.
#
# All four thresholds are HEURISTICS fitted to a handful of clips, not statistics.
# Re-calibrate them against the SNU lab force-plate data (that is the whole point
# of the validation study) and change them here — nowhere else reads raw numbers.
# --------------------------------------------------------------------------- #
G_STD = 9.81            # m/s²
G_TOL = 1.5             # m/s² → accept g ∈ [8.31, 11.31] (≈ ±15% scale error)
POSE_DET_MIN = 90.0     # % of frames with a pose, whole clip
WINDOW_DET_MIN = 85.0   # % of frames with a pose, take-off→landing window
SCALE_CV_MAX = 0.25     # limb-length scale spread; above this the scale is guesswork
SLOWMO_R2_MIN = 0.95    # vertical-parabola fit quality required to trust the slow-mo k

# Metrics whose VALUE is proportional to the pixel→meter scale. When the scale is
# unverified these must not be quoted; every other metric (angles, times, frames)
# is scale-invariant and stays. Keep this set in sync with any new metric.
ABSOLUTE_METRIC_KEYS = frozenset({
    "takeoff_speed_mps", "takeoff_vx_mps", "takeoff_vy_mps",
    "estimated_distance_com_m", "estimated_distance_foot_m",
    "apex_com_height_m", "landing_com_forward_drift_m", "landing_com_vs_foot_m",
})


def _finite(x) -> bool:
    """True only for a real, finite number. None / NaN / inf / non-numeric → False."""
    return isinstance(x, (int, float)) and not isinstance(x, bool) and math.isfinite(x)


def compute_reliability(*, det_pct, window_det_pct, scale_cv, scale_is_manual: bool,
                        g_implied, slowmo_detected: bool, slowmo_fit_r2,
                        events_reliable: bool) -> dict:
    """Decide (a) whether the landmarks were really tracked and (b) whether absolute
    lengths/speeds may be quoted. Pure function — no I/O, no globals, so the gates are
    unit-testable without a video. Any non-finite input fails closed (unreliable).

    scale_is_manual is a BOOLEAN on purpose: keying off a display string like
    "manual" lets a typo or a new scale source silently fall through to "trusted".
    """
    scale_is_manual = bool(scale_is_manual)

    # (a) tracking — did we actually see the body, over the whole clip AND in the
    # take-off→landing window (a 95% whole-clip rate can still hide a broken flight).
    cv_ok = True if scale_is_manual else (_finite(scale_cv) and scale_cv <= SCALE_CV_MAX)
    tracking_reliable = bool(
        _finite(det_pct) and det_pct >= POSE_DET_MIN
        and _finite(window_det_pct) and window_det_pct >= WINDOW_DET_MIN
        and cv_ok
    )

    # (b) absolute scale — see the identifiability note above.
    if not (tracking_reliable and bool(events_reliable)):
        absolute_ok = False
        if not tracking_reliable:
            reason = "몸 추적이 불안정해요"
        else:
            reason = "도약·착지 검출이 불안정해요"
    elif slowmo_detected:
        # k was DERIVED from g, so g is no longer an independent check of the scale.
        # Only a user-asserted scale reduces this to one unknown — and even then the
        # numbers inherit whatever error that asserted scale carries. Say so.
        absolute_ok = scale_is_manual and _finite(slowmo_fit_r2) and slowmo_fit_r2 >= SLOWMO_R2_MIN
        reason = ("슬로우모션 보정 — 사용자가 지정한 스케일에 의존" if absolute_ok
                  else "슬로우모션이라 크기 기준을 검증할 수 없어요")
    else:
        absolute_ok = _finite(g_implied) and abs(g_implied - G_STD) <= G_TOL
        reason = ("중력 검증 통과" if absolute_ok else
                  (f"중력 검증 {g_implied:.2f} m/s² (기준 {G_STD})" if _finite(g_implied)
                   else "중력 검증값을 구하지 못했어요"))

    return {
        "tracking_reliable": tracking_reliable,
        "absolute_metrics_reliable": bool(absolute_ok),
        "reliability_reason": reason,
    }


def scale_from_gravity(scale, g_implied):
    """The scale the flight parabola implies, given g must be 9.81. Diagnostic ONLY —
    nothing downstream consumes it, because we cannot tell a scale error from an fps
    error (see the identifiability note). Returns None when it is not computable."""
    if not (_finite(scale) and scale > 0 and _finite(g_implied) and g_implied > 0):
        return None
    return scale * g_implied / G_STD

def coaching(m: dict) -> list[str]:
    """Phase-by-phase, data-grounded cues. Each cue cites the measured value, the plot
    panel that shows it (그래프 1~4), and a concrete fix, in jump order:
    반동 → 이지(속도·신전·각도) → 체공(팔) → 착지. A lead cue names the single biggest
    fixable thing. Honest per the research: speed is the dominant lever; optima are flat
    bands (not single ideals); extension/arm are directional, not precise angle targets.
    Plot panel map: 1=관절각, 2=팔 높이, 3=CoM 궤적, 4=CoM 변위."""
    def g(k):
        v = m.get(k)
        return v if isinstance(v, (int, float)) and math.isfinite(v) else None

    cues = []
    fixes = []   # (priority, short_label); lower priority number = fix first

    # If the body was never reliably found, NOTHING downstream is trustworthy — not even
    # the angles. Checked before the event gate because bad landmarks also fake bad events.
    if not m.get("tracking_reliable", True):
        det = m.get("pose_detection_pct")
        det_txt = f"(검출률 {det:.0f}%) " if _finite(det) else ""
        return [f"⚠️ 영상에서 몸을 안정적으로 찾지 못했어요{det_txt}— 지표를 신뢰할 수 없습니다. "
                "측면에서 전신이 프레임 안에 끝까지 들어오게, 밝은 곳에서 카메라를 고정하고 다시 촬영해 주세요."]

    # If the jump events were mis-detected (impossible implied gravity), every event-windowed
    # cue would be garbage. Say so honestly instead of listing confident-but-wrong advice.
    if not m.get("events_reliable", True):
        return ["⚠️ 도약·착지 순간 검출이 불안정해 대부분의 지표를 신뢰할 수 없어요. 측면에서 전신이 끝까지 보이게, 카메라를 고정하고 다시 촬영해 분석해 주세요."]

    # Scale-dependent numbers (speed, distance, heights) are only quoted when the pixel→meter
    # scale has an independent check. Gated at GENERATION, not by filtering finished strings —
    # a string filter would also eat the "9.81 m/s²" inside the explanation itself.
    abs_ok = bool(m.get("absolute_metrics_reliable", True))

    cm = g("countermovement_knee_min_deg")
    sp, vx, vy = g("takeoff_speed_mps"), g("takeoff_vx_mps"), g("takeoff_vy_mps")
    knee_to, hip_to, ank_to = g("knee_angle_at_takeoff_deg"), g("hip_angle_at_takeoff_deg"), g("ankle_angle_at_takeoff_deg")
    a = g("takeoff_angle_deg")
    arm = g("arm_swing_range_deg")
    lk = g("landing_knee_min_deg")

    # 반동(load) — 그래프 1, 이지 직전 무릎 골짜기. 추적·구간이 불충분하면 단정하지 않는다.
    if cm is not None and m.get("countermovement_reliable", True):
        if cm > 120:
            cues.append(f"👉 【반동】 무릎이 최저 {cm:.0f}°까지만 굽혀졌어요(그래프 1, 이지 직전 골짜기). 반동이 얕습니다 — 조금 더 앉아 신장-단축 주기를 살리면 폭발력이 커집니다.")
            fixes.append((3, "반동 더 깊게 앉기"))
        else:
            cues.append(f"【반동】 무릎을 최저 {cm:.0f}°까지 충분히 굽혔어요(그래프 1). 반동 깊이는 좋습니다.")

    # 이지·속도 — 그래프 4 변위 추세, 거리의 최우선 지렛대.
    # 문헌상 거리를 지배하는 변수라 이걸 못 말하는 게 이 게이트의 실질 손실이다.
    if sp is not None and abs_ok:
        comp = f"(수평 {vx:.1f}·수직 {vy:.1f} m/s) " if (vx is not None and vy is not None) else ""
        slo, shi = REF["takeoff_speed_mps"]
        if sp < slo:
            cues.append(f"👉 【이지·속도】 이지 속도 {sp:.2f} m/s {comp}— 기준 {slo}–{shi}보다 낮아요(그래프 4의 CoM 변위 추세가 완만). 거리의 가장 큰 지렛대예요(거리 ∝ 속도²). 반동 후 폭발적으로 지면을 밀어내는 데 집중하세요.")
            fixes.append((1, "이지 속도 높이기 — 거리의 가장 큰 지렛대"))
        else:
            cues.append(f"【이지·속도】 이지 속도 {sp:.2f} m/s {comp}— 양호({slo}–{shi}, 그래프 4). 거리의 핵심 지렛대이니 유지·향상하세요.")

    # 이지·삼중신전 — 그래프 1 초록 점선(takeoff)
    joints = "·".join(p for p in (
        f"무릎 {knee_to:.0f}°" if knee_to is not None else None,
        f"고관절 {hip_to:.0f}°" if hip_to is not None else None,
        f"발목 {ank_to:.0f}°" if ank_to is not None else None) if p)
    ext_pair = [v for v in (knee_to, hip_to) if v is not None]
    if joints:
        if ext_pair and min(ext_pair) < 160:
            low = "무릎" if (knee_to is not None and (hip_to is None or knee_to <= hip_to)) else "고관절"
            cues.append(f"👉 【이지·신전】 이지 순간 {joints}(그래프 1, 초록 점선). {low}이 덜 펴졌어요 — 끝까지 밀어 신전을 키우면 이지 속도가 올라갑니다(완전 락아웃을 억지로 만들 필요는 없어요).")
            fixes.append((2, f"{low} 끝까지 신전하기"))
        else:
            cues.append(f"【이지·신전】 이지 순간 {joints}(그래프 1, 초록 점선). 충분히 신전됐어요.")

    # 이지·각도 — 그래프 3 궤적 초기 기울기, 평탄한 최적
    if a is not None:
        glo, ghi = REF["takeoff_angle_deg"]
        if a < glo:
            cues.append(f"【이지·각도】 이지 각도 {a:.0f}°(그래프 3, 궤적 초기 기울기) — 다소 낮아 체공이 짧아요. 수직 성분을 약간 더(약 20~40°). 단 각도 최적은 평탄하니 속도부터.")
        elif a > ghi:
            cues.append(f"👉 【이지·각도】 이지 각도 {a:.0f}°(그래프 3) — 다소 높아요. 위로 과하게 뛰면 수평 속도를 잃어요. 앞으로 더 밀어내세요(단 ±5° 손실은 작으니 속도가 우선).")
            fixes.append((5, "각도 낮춰 앞으로 더 밀기"))
        else:
            cues.append(f"【이지·각도】 이지 각도 {a:.0f}°(그래프 3) — 양호(약 20~40°, 평탄한 최적).")

    # 체공·팔 스윙 — 그래프 2. 90°는 채점 밴드와 동일 기준(휴리스틱). 추적 불충분하면 생략.
    if arm is not None and m.get("arm_swing_reliable", True):
        if arm < REF["arm_swing_min_deg"]:
            cues.append(f"👉 【체공·팔】 팔 스윙 범위 {arm:.0f}°(그래프 2)로 작아요. 팔을 크게 뒤→앞·위로, 이지 직전 어깨를 감속시켜 동기화하면(크기·타이밍 둘 다) 이지 속도가 올라 거리가 늘어요.")
            fixes.append((4, "팔 더 크게·동기화해 스윙하기"))
        else:
            cues.append(f"【체공·팔】 팔 스윙 범위 {arm:.0f}°(그래프 2) — 의미 있는 스윙이에요. 좋습니다(타이밍도 함께 신경쓰면 더 좋아요).")

    # 착지 — 핵심은 '무게중심이 발 앞으로 빠졌나'(뒤로 앉으면 거리 손실). 그래프 3(CoM 궤적)
    # 무릎 깊이는 흡수(안전) 신호일 뿐 점수 핵심이 아니다.
    cvf = g("landing_com_vs_foot_m")            # +면 무게중심이 발자국보다 앞
    reliable = bool(m.get("landing_metric_reliable"))
    kl = g("knee_angle_at_landing_deg")          # 닿는 순간 무릎(다리 앞으로 뻗었나)
    if reliable and cvf is not None:
        # 무게중심-발 거리는 절대 길이라 스케일 검증이 있어야 인용한다. 각도 기반 보조 경고는 무관.
        if abs_ok:
            if cvf < 0:
                cues.append(f"👉 【착지】 착지 후 무게중심이 발자국보다 {abs(cvf):.2f}m 뒤에 남았어요(그래프 3 — 뒤로 앉음). 거리는 가장 뒤 접지점으로 재니 이만큼 손해예요. 닿는 순간 팔·상체를 앞으로 던지고 엉덩이를 발 앞으로 통과시키세요.")
                fixes.append((2, "착지에서 무게중심을 발 앞으로 보내기"))
            else:
                cues.append(f"【착지】 착지 후 무게중심이 발자국보다 {cvf:.2f}m 앞으로 빠졌어요(그래프 3). 뒤로 안 앉아 거리 손실이 없습니다 — 좋습니다.")
        # 닿는 순간 다리가 너무 뻣뻣하고(>160°) 흡수도 없으면(무릎 최저>150°) stiff-leg 보조 경고
        if kl is not None and kl > 160 and (lk is None or lk > 150):
            cues.append(f"【착지·보조】 닿는 순간 다리가 {kl:.0f}°로 뻣뻣했어요(stiff-leg). 무릎으로 충격을 흡수하면 더 안전합니다.")
    else:
        # 추적이 불안정하면 착지 평가를 보류한다. 이때 무릎 최저각(lk)도 신뢰할 수 없으니 수치를 인용하지 않는다.
        cues.append("【착지】 착지 추적이 불안정해 착지 평가를 보류했어요. 측면에서 발이 끝까지 보이게 고정 촬영하면 정확해집니다.")

    # lead cue — 가장 먼저 고칠 한 가지
    if fixes:
        fixes.sort(key=lambda x: x[0])
        cues.insert(0, f"👉 가장 먼저 고칠 것: {fixes[0][1]}.")
    else:
        cues.insert(0, "👍 전반적으로 균형 잡힌 도약이에요 — 이제 이지 속도를 더 끌어올리면 거리가 가장 크게 늘어요.")

    # 절대 수치를 숨겼다면 왜 숨겼는지, 어떻게 하면 볼 수 있는지 맨 앞에 밝힌다.
    if not abs_ok:
        why = m.get("reliability_reason") or "크기 기준을 검증할 수 없어요"
        cues.insert(0, f"📏 화면 속 크기 기준을 확인하지 못해({why}) 속도·거리·높이 수치는 숨겼어요. "
                       "각도와 타이밍은 크기와 무관하므로 아래 내용은 그대로 유효합니다. "
                       "정확한 거리를 보려면 화면 안에 길이를 아는 물체(1m 자 등)를 함께 찍거나 --scale-px-per-m 을 지정해 주세요.")
    return cues

def photo_feedback(m: dict) -> dict:
    """Per-photo good/bad feedback for the 3 key-moment screenshots (도약·뛰는 중간·착지),
    grounded ONLY in already-measured metrics (same thresholds as coaching()/REF). Each metric
    is read independently, so partial data still yields what it can. Returns
    {phase: {"good": [...], "bad": [...]}}."""
    def g(k):
        v = m.get(k)
        return v if isinstance(v, (int, float)) and math.isfinite(v) else None

    slo, _ = REF["takeoff_speed_mps"]
    fb = {"takeoff": {"good": [], "bad": []}, "mid": {"good": [], "bad": []}, "landing": {"good": [], "bad": []}}
    # Bad landmarks fake everything downstream, so a tracking failure empties all three buckets
    # (the UI then shows "분석 데이터가 부족해요" rather than confident-but-wrong captions).
    if not m.get("tracking_reliable", True):
        return fb
    ev_ok = bool(m.get("events_reliable", True))   # takeoff/mid feedback depends on correct event frames
    abs_ok = bool(m.get("absolute_metrics_reliable", True))   # scale-dependent numbers gated at generation

    # 도약(takeoff): 삼중신전 · 각도 · 속도
    knee_to, hip_to = g("knee_angle_at_takeoff_deg"), g("hip_angle_at_takeoff_deg")
    ext = [v for v in (knee_to, hip_to) if v is not None]
    a, sp = g("takeoff_angle_deg"), g("takeoff_speed_mps")
    t = fb["takeoff"]
    if ext and ev_ok:
        if min(ext) >= 160:
            parts = "·".join(p for p in (f"무릎 {knee_to:.0f}°" if knee_to is not None else None,
                                         f"고관절 {hip_to:.0f}°" if hip_to is not None else None) if p)
            # Describe what was measured; do NOT claim propulsion from a static angle — propulsion is
            # driven by extension VELOCITY/timing (not captured by a single-frame angle). See compare._ext_score.
            t["good"].append(f"이지 순간 {parts}로 충분히 신전됐어요(추진은 신전 '속도'가 좌우 — 각도는 참고).")
        else:
            low = "무릎" if (knee_to is not None and (hip_to is None or knee_to <= hip_to)) else "고관절"
            t["bad"].append(f"{low}이 덜 펴졌어요 — 끝까지 밀어 신전을 키우세요(이지 속도 ↑).")
    if a is not None and ev_ok:
        if a > 40:
            t["bad"].append(f"도약 각도가 {a:.0f}°로 높아요 — 위로 과하게 떠 수평 속도를 잃어요.")
        elif a < 20:
            t["bad"].append(f"도약 각도가 {a:.0f}°로 낮아요 — 체공이 짧아요(약 20~40°).")
        else:
            t["good"].append(f"도약 각도 {a:.0f}°로 적절해요(평탄한 최적 구간).")
    if sp is not None and ev_ok and abs_ok:
        if sp < slo:
            t["bad"].append(f"이지 속도 {sp:.2f} m/s로 낮아요 — 더 폭발적으로 미세요(거리의 최대 지렛대).")
        else:
            t["good"].append(f"이지 속도 {sp:.2f} m/s로 지면을 잘 밀어냈어요.")

    # 뛰는 중간(flight): 팔 스윙은 도약~체공 시퀀스 신호. 크기·타이밍 둘 다 중요(타이밍은 점수엔 미반영).
    arm = g("arm_swing_range_deg")
    mid = fb["mid"]
    if arm is not None and m.get("arm_swing_reliable", True):
        if arm >= REF["arm_swing_min_deg"]:        # 90°: 채점 밴드와 동일 기준
